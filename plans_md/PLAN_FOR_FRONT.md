```
│   │   ├── Modal.tsx               # Модальное окно

│   ├── useGameLoop.ts              # Хук игрового цикла
│   ├── useSnake.ts                 # Хук управления змейкой
│   ├── useCollision.ts             # Хук проверки коллизий
│   └── useTelegram.ts              # Хук интеграции с Telegram
├── services/
│   ├── api.ts                      # API клиент для бэкенда
│   ├── gameEngine.ts               # Игровая логика
│   └── levelManager.ts             # Управление уровнями
├── types/
│   ├── game.ts                     # Типы для игры
│   └── api.ts                      # Типы для API
├── constants/
│   ├── game.ts                     # Константы игры
│   └── levels.ts                   # Конфигурации уровней
└── utils/
    ├── collision.ts                # Утилиты для коллизий
    └── helpers.ts                  # Общие утилиты
```

### 1.2. Состояние приложения
- **Zustand store** для управления глобальным состоянием
- **Локальное состояние** компонентов для UI элементов
- **Refs** для игрового цикла и анимаций

## 2. Игровая механика

### 2.1. Основные элементы
- **Змейка**: массив координат, голова + тело
- **Еда**: случайная позиция на поле
- **Препятствия**: статические и динамические объекты
- **Поле**: сетка с границами

### 2.2. Управление
- **Клавиатура**: стрелки, WASD
- **Свайпы**: для мобильных устройств
- **Telegram кнопки**: MainButton для паузы/продолжения

### 2.3. Логика игры
- Движение змейки с постоянной скоростью
- Рост при поедании еды
- Смерть при столкновении с собой, препятствиями или границами
- Подсчет очков (за еду + бонусы за скорость)

## 3. Система уровней

### 3.1. Конфигурация уровня
```typescript
interface Level {
  id: string;
  name: string;
  width: number;
  height: number;
  obstacles: Obstacle[];
  foodTypes: FoodType[];
  speed: number;
  maxScore: number;
  isUnlocked: boolean;
}
```

### 3.2. Типы препятствий
- **Стены**: статические блоки
- **Лабиринт**: сложные проходы
- **Движущиеся препятствия**: патрулирующие объекты
- **Телепорты**: переходы между частями поля
- **Ловушки**: временные опасные зоны

### 3.3. Типы еды
- **Обычная**: +1 к длине, +10 очков
- **Золотая**: +2 к длине, +50 очков
- **Скоростная**: временное ускорение
- **Замедляющая**: временное замедление
- **Бонусная**: специальные эффекты
